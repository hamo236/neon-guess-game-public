'FINAL' : match.matchId === 'consolation' ? '3RD PLACE MATCH' : 'SEMIFINAL'} · ROUND {match.roundNumber}</p><h2 className="font-display-lg text-display-lg text-white">{state.players[actions.playerId]?.name} <span className="text-secondary-fixed">VS</span> {state.players[opponentId]?.name}</h2></div><Timer endTimestamp={match.roundEndTimestamp} /></div><div className="glass-panel rounded-xl p-4 flex items-center justify-between"><div><div className="font-label-caps text-label-caps text-on-surface-variant">CATEGORY</div><div className="text-white mt-1">{state.category}</div></div><div className="text-right"><div className="font-label-caps text-label-caps text-on-surface-variant">SCORE</div><div className="font-stats-num text-stats-num text-primary-fixed">{match.scores?.[actions.playerId] ?? 0} — {match.scores?.[opponentId] ?? 0}</div></div></div><TargetCard target={target} ready={actions.targetReady} /><section className="glass-panel-heavy rounded-2xl p-4 sm:p-6 space-y-4 border border-secondary-fixed/20 shadow-[0_18px_60px_rgba(0,0,0,0.2)]"><div className="flex items-center justify-between"><div><div className="font-label-caps text-label-caps text-secondary-fixed">GUESS BOARD</div><p className="text-sm text-on-surface-variant mt-1">Find the target {state.players[opponentId]?.name} is protecting.</p></div><span className="material-symbols-outlined text-secondary-fixed">touch_app</span></div><GuessGrid items={items} guessedTargetId={guess?.targetId} onGuess={actions.recordGuess} locked={!!guess || seconds === 0 || !actions.targetReady || !actions.canMutateCompetitive} /><p className="text-center text-sm text-on-surface-variant">{guess ? (guess.correct ? 'Correct guess recorded.' : 'Guess recorded. Waiting for the synchronized match result.') : 'Choose one card. The synchronized timer resolves the match for every player.'}</p></section></div>; }
