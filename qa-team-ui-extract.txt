function TeamBattleGameplay({ state, actions }) { const myTeam = Object.values(state.teams || {}).find((team) => team.playerIds.includes(actions.playerId)); const myLabel = myTeam?.teamId === 'team_a' ? 'TEAM A' : 'TEAM B'; const opponentLabel = myLabel === 'TEAM A' ? 'TEAM B' : 'TEAM A'; const persistedConfirmationTeams = Object.values(state.match?.confirmations || {}).filter((entries) => Object.values(entries || {}).some((entry) => entry?.roundNumber === state.roundNumber && entry?.matchId === state.match?.matchId)).map((entries) => Object.values(entries || {})[0]?.teamId).filter(Boolean); const lockedConfirmationTeam = state.match?.confirmationTeamId || state.match?.confirmationTeamIds?.[0] || persistedConfirmationTeams[0] || null; const myTeamRequired = Boolean(lockedConfirmationTeam && myTeam?.teamId === lockedConfirmationTeam); const confirmations = state.match?.confirmations?.[myTeam?.teamId] || {}; const confirmedIds = myTeam?.playerIds.filter((id) => confirmations[id]?.roundNumber === state.roundNumber && confirmations[id]?.matchId === state.match?.matchId) || []; const hasConfirmed = confirmedIds.includes(actions.playerId); const requiredTeamsComplete = Boolean(lockedConfirmationTeam && myTeamRequired && myTeam?.playerIds?.length === 2 && myTeam.playerIds.every((id) => confirmations[id]?.roundNumber === state.roundNumber && confirmations[id]?.matchId === state.match?.matchId)); useEffect(() => { if (requiredTeamsComplete && state.hostId === actions.playerId) actions.resolveTeamRound(); }, [requiredTeamsComplete, state.hostId, actions]); return <div className="space-y-5"><div className="rounded-3xl border border-white/10 bg-gradient-to-r from-white/10 via-white/5 to-transparent p-4 sm:p-5 shadow-[0_12px_36px_rgba(0,0,0,0.16)]"><p className="font-label-caps text-label-caps text-primary-fixed">2v2 TEAM BATTLE آ· ROUND {state.roundNumber} / 3</p><h2 className="font-display-lg text-display-lg text-white mt-1">{state.category}</h2></div><TeamBattleIdentity state={state} actions={actions} /><TargetCard target={actions.privateTarget} ready={actions.targetReady} mode="opponent" opponentTeamLabel={opponentLabel} /><section className="glass-panel-heavy rounded-3xl p-4 sm:p-5 border border-secondary-fixed/35 bg-gradient-to-br from-secondary-fixed/15 via-secondary-fixed/5 to-transparent shadow-[0_20px_70px_rgba(0,0,0,0.24)]" aria-label="Team confirmation"><div className="flex items-center justify-between gap-3"><span className="font-label-caps text-label-caps text-secondary-fixed">{opponentLabel} GUESSED CORRECT</span><span className="text-xs text-on-surface-variant">{confirmedIds.length}/2 CONFIRMED</span></div><button type="button" disabled={!myTeamRequired || hasConfirmed || !actions.canMutateCompetitive} onClick={() => actions.confirmTeamGuess()} className={`touch-feedback mt-4 min-h-14 w-full rounded-2xl px-4 py-4 font-headline-sm transition-all active:scale-[0.99] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-secondary-fixed disabled:cursor-not-allowed ${hasConfirmed ? 'border border-primary-fixed/50 bg-primary-fixed/10 text-primary-fixed' : 'bg-secondary-fixed text-on-secondary-fixed hover:brightness-110 disabled:opacity-40'}`}><span className="material-symbols-outlined align-middle mr-2" aria-hidden="true">{hasConfirmed ? 'verified' : 'check_circle'}</span>{hasConfirmed ? `CONFIRMED آ· ${confirmedIds.length}/2` : `${opponentLabel} GUESSED CORRECT`}</button></section></div>; }
function BracketMatch({ match, state, active }) { return <div className={`glass-panel rounded-xl p-4 border ${active ? 'border-primary-fixed neon-border' : 'border-white/10'}`}><div className="flex items-center justify-between gap-2"><span className="font-label-caps text-label-caps text-primary-fixed">{match.matchId === 'semi_a' ? 'SEMIFINAL A' : match.matchId === 'semi_b' ? 'SEMIFINAL B' : match.matchId === 'consolation' ? '3RD PLACE' : 'FINAL'}</span><span className="text-[10px] text-on-surface-variant">{match.status}</span></div><div className="mt-3 space-y-2">{(match.playerIds.length ? match.playerIds : ['TBD', 'TBD']).map((id, index) => <div key={`${id}-${index}`} className="flex items-center justify-between rounded bg-white/5 px-3 py-2 text-sm"><span className={match.result?.winnerId === id ? 'text-primary-fixed font-semibold' : 'text-on-surface'}>{state.players[id]?.name || id}</span><span className="text-on-surface-variant">{match.scores?.[id] ?? 0}</span></div>)}</div></div>; }
function TournamentTransition({ state, actions }) { const seconds = useCountdown(state.transitionEndTimestamp); useEffect(() => { if (seconds === 0 && state.hostId === actions.playerId) actions.advanceTournament(); }, [seconds, state.hostId, actions]); return <section className="glass-panel-heavy rounded-xl p-6 sm:p-10 text-center space-y-6"><p className="font-label-caps text-label-caps text-secondary-fixed">SEMIFINALS COMPLETE · ALL FOUR PLAYERS</p><h2 className="font-display-lg text-display-lg text-white">NEXT MATCHES</h2><div className="grid sm:grid-cols-2 gap-4 text-left"><div className="glass-panel rounded-lg p-4 border border-primary-fixed/30"><div className="font-label-caps text-label-caps text-primary-fixed">FINAL · WINNERS</div>{state.matches.final.playerIds.map((id) => <div key={id} className="text-white mt-2">{state.players[id]?.name}</div>)}</div><div className="glass-panel rounded-lg p-4 border border-secondary-fixed/30"><div className="font-label-caps text-label-caps text-secondary-fixed">3RD PLACE · LOSERS</div>{state.matches.consolation.playerIds.map((id) => <div key={id} className="text-white mt-2">{state.players[id]?.name}</div>)}</div></div><div><div className="font-label-caps text-label-caps text-primary-fixed">NEXT MATCH STARTING IN</div><div className="font-display-lg text-display-lg text-white mt-1">{seconds > 0 ? seconds : 'STARTING'}</div></div>{state.hostId !== actions.playerId && <p className="text-xs text-on-surface-variant">The bracket is locked. Waiting for the host’s synchronized transition.</p>}</section>; }
function TournamentFinal({ state, actions }) { const placements = [['CHAMPION', state.winnerId], ['2ND PLACE', state.secondPlaceId], ['3RD PLACE', state.thirdPlaceId], ['4TH PLACE', state.fourthPlaceId]]; return <section className="glass-panel-heavy rounded-xl p-6 sm:p-10 text-center space-y-6"><div className="text-5xl">🏆</div><div><p className="font-label-caps text-label-caps text-secondary-fixed">TOURNAMENT COMPLETE · FOUR PLAYER RESULTS</p><h2 className="font-display-lg text-display-lg text-white">FINAL RESULTS</h2></div><div className="grid sm:grid-cols-2 gap-3 text-left">{placements.map(([label, id]) => { const stats = state.playerStats?.[id]; const reward = state.rewards?.[id]; return <div key={label} className={`glass-panel rounded-lg p-4 ${label === 'CHAMPION' ? 'border border-primary-fixed/60' : 'border border-white/10'}`}><div className="font-label-caps text-label-caps text-primary-fixed">{label}</div><div className="text-white text-lg mt-2">{id ? state.players[id]?.name : 'Awaiting result'}</div><div className="text-xs text-on-surface-variant mt-1">Score {stats?.score || 0} · Correct {stats?.correctGuesses || 0}</div>{reward && <div className="text-xs text-secondary-fixed mt-1">Reward +{reward.points}</div>}</div>; })}</div><button type="button" onClick={actions.leave} className="touch-feedback min-h-11 w-full py-3 rounded-lg border border-white/20 text-on-surface">Return to Lobby</button></section>; }
function TournamentMatchResult({ state, match }) { const winnerId = match.result?.winnerId; const loserId = match.result?.loserId; return <section className="glass-panel-heavy rounded-xl p-6 text-center space-y-5"><p className="font-label-caps text-label-caps text-secondary-fixed">{match.matchId === 'final' ? 'FINAL MATCH COMPLETE' : match.matchId === 'consolation' ? '3RD PLACE MATCH COMPLETE' : 'SEMIFINAL COMPLETE'}</p><h2 className="font-display-lg text-display-lg text-white">{state.players[winnerId]?.name} ADVANCES</h2><div className="grid sm:grid-cols-2 gap-3 text-left">{match.playerIds.map((id) => <div key={id} className="glass-panel rounded-lg p-4 border border-white/10"><div className="text-white text-lg">{state.players[id]?.name}</div><div className="text-xs text-on-surface-variant mt-1">Score {match.result?.scores?.[id] || 0} · {match.result?.guesses?.[id]?.correct ? 'Correct' : 'Miss'}</div></div>)}</div><div className="text-sm text-on-surface-variant">{state.players[loserId]?.name} records the match result. Waiting for the Firebase-authoritative bracket transition.</div></section>; }
function TournamentGameplay({ state, actions, match }) { const items = useMemo(() => getItemsByCategory(state.category), [state.category]); const target = actions.privateTarget; const guess = match.guesses?.[actions.playerId]; const seconds = useCountdown(match.roundEndTimestamp); useEffect(() => { if (seconds === 0 && state.hostId === actions.playerId) actions.resolveTournamentMatch(match.matchId); }, [seconds, state.hostId, actions, match.matchId]); const opponentId = match.playerIds.find((id) => id !== actions.playerId); return <div className="space-y-5"><div className="flex flex-wrap items-center justify-between gap-3"><div><p className="font-label-caps text-label-caps text-primary-fixed">{match.matchId === 'final' ? 'FINAL' : match.matchId === 'consolation' ? '3RD PLACE MATCH' : 'SEMIFINAL'} · ROUND {match.roundNumber}</p><h2 className="font-display-lg text-display-lg text-white">{state.players[actions.playerId]?.name} <span className="text-secondary-fixed">VS</span> {state.players[opponentId]?.name}</h2></div><Timer endTimestamp={match.roundEndTimestamp} /></div><div className="glass-panel rounded-xl p-4 flex items-center justify-between"><div><div className="font-label-caps text-label-caps text-on-surface-variant">CATEGORY</div><div className="text-white mt-1">{state.category}</div></div><div className="text-right"><div className="font-label-caps text-label-caps text-on-surface-variant">SCORE</div><div className="font-stats-num text-stats-num text-primary-fixed">{match.scores?.[actions.playerId] ?? 0} — {match.scores?.[opponentId] ?? 0}</div></div></div><TargetCard target={target} ready={actions.targetReady} /><section className="glass-panel-heavy rounded-2xl p-4 sm:p-6 space-y-4 border border-secondary-fixed/20 shadow-[0_18px_60px_rgba(0,0,0,0.2)]"><div className="flex items-center justify-between"><div><div className="font-label-caps text-label-caps text-secondary-fixed">GUESS BOARD</div><p className="text-sm text-on-surface-variant mt-1">Find the target {state.players[opponentId]?.name} is protecting.</p></div><span className="material-symbols-outlined text-secondary-fixed">touch_app</span></div><GuessGrid items={items} guessedTargetId={guess?.targetId} onGuess={actions.recordGuess} locked={!!guess || seconds === 0 || !actions.targetReady || !actions.canMutateCompetitive} /><p className="text-center text-sm text-on-surface-variant">{guess ? (guess.correct ? 'Correct guess recorded.' : 'Guess recorded. Waiting for the synchronized match result.') : 'Choose one card. The synchronized timer resolves the match for every player.'}</p></section></div>; }
function TournamentBoard({ state, actions }) { const activeMatch = Object.values(state.matches || {}).find((match) => match.status === 'playing' && match.playerIds.includes(actions.playerId)); if (state.phase === MODE_PHASES.TRANSITION) return <TournamentTransition state={state} actions={actions} />; if (state.phase === MODE_PHASES.RESULTS && state.status === 'finished') return <TournamentFinal state={state} actions={actions} />; const completedMatch = Object.values(state.matches || {}).find((match) => match.status === 'finished' && match.playerIds.includes(actions.playerId) && match.result); const finishedSemis = [state.matches.semi_a, state.matches.semi_b].every((match) => match.status === 'finished'); return <div className="space-y-6"><div className="grid sm:grid-cols-2 gap-4"><BracketMatch match={state.matches.semi_a} state={state} active={activeMatch?.matchId === 'semi_a'} /><BracketMatch match={state.matches.semi_b} state={state} active={activeMatch?.matchId === 'semi_b'} /></div><div className="flex justify-center text-secondary-fixed font-label-caps">WINNERS ADVANCE · LOSERS PLAY FOR 3RD</div><div className="grid sm:grid-cols-2 gap-4"><BracketMatch match={state.matches.final} state={state} active={activeMatch?.matchId === 'final'} /><BracketMatch match={state.matches.consolation} state={state} active={activeMatch?.matchId === 'consolation'} /></div>{activeMatch ? <TournamentGameplay state={state} actions={actions} match={activeMatch} /> : completedMatch ? <TournamentMatchResult state={state} match={completedMatch} /> : finishedSemis ? <TournamentTransition state={state} actions={actions} /> : <div className="glass-panel rounded-xl p-5 text-center text-on-surface-variant">You are waiting for your assigned match to begin. The bracket is synchronized by Firebase.</div>}</div>; }
